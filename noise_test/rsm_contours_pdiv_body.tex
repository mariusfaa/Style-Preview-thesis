\pgfplotsset{
    pdivPanel/.style={
        scale only axis=true,
        width=\panelW,
        height=\panelH,
        xmin=-1, xmax=1,
        ymin=-1, ymax=1,
        tick label style={font=\scriptsize},
        label style={font=\footnotesize},
        xlabel near ticks,
        ylabel near ticks,
        x label style={yshift=4pt},
        y label style={yshift=-8pt},
        table/col sep=comma,
        point meta min=0,
        point meta max=1,
        colormap/viridis,
        axis on top,
        colorbar,
        colorbar style={
            width=0.18cm,
            title={$P_{\mathrm{div}}$},
            title style={font=\tiny, yshift=-2pt},
            ytick={0,0.25,0.5,0.75,1},
            tick label style={font=\tiny},
            tick align=outside
        }
    },
    pdivSurface/.style={
        matrix plot*,
        mesh/cols=40,
        point meta=explicit,
        shader=interp
    },
    designMarks/.style={
        only marks,
        mark=*,
        mark size=1.35pt,
        mark options={fill=white, draw=black, line width=0.4pt}
    },
    pdivBoundary/.style={
        boundaryRed,
        line width=1.1pt,
        unbounded coords=jump
    }
}
