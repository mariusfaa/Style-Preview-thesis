% --- Load metadata so axis ranges / labels are data-driven ---
\pgfplotstableread[col sep=comma]{meta.csv}\metatable

\pgfplotstablegetelem{1}{value}\of\metatable \let\controllername\pgfplotsretval

% Bar-chart x range: 0.5 ... numdirs + 0.5
\pgfmathsetmacro{\xmaxbar}{\numdirs + 0.5}

% Direction lookup table with derived spherical coordinates
\pgfplotstableread[col sep=comma]{impulse_directions.csv}\dirtable
\pgfplotstablecreatecol[create col/expr={\pgfplotstablerow+1}]{idx}{\dirtable}
\pgfplotstablecreatecol[create col/expr={atan2(\thisrow{dy},\thisrow{dx})}]{azimuth}{\dirtable}
\pgfplotstablecreatecol[
    create col/expr={atan2(\thisrow{dz},sqrt(\thisrow{dx}*\thisrow{dx}+\thisrow{dy}*\thisrow{dy}))}
]{elevation}{\dirtable}
